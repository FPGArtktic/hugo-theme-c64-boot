{{- /* dir-entries.html — turn pages into directory entries, and return the slice. Block counts are real: a 1541 stores 254 data bytes per block, so a file is ceil(len(content) / 254) blocks and never fewer than one. Nothing here is invented, which is the whole reason the format is worth borrowing. Context: a dict with pages the pages to list current the page being read, for aria-current */ -}} {{- $current := .current -}} {{- $entries := slice -}} {{- range .pages -}} {{- $blocks := math.Ceil (div (len .RawContent | float) 254.0) | int -}} {{- if lt $blocks 1 }}{{ $blocks = 1 }}{{ end -}} {{- $type := "PRG" -}} {{- if .Draft -}} {{- $type = "DEL" -}} {{- else if or (eq .Kind "taxonomy") (eq .Kind "term") -}} {{- $type = "REL" -}} {{- else if and (eq .Kind "page") .Section -}} {{- $type = "SEQ" -}} {{- end -}} {{- $entries = $entries | append (dict "name" (upper (substr (or .LinkTitle .Title) 0 16)) "url" .RelPermalink "blocks" $blocks "type" $type "date" (cond (eq $type "SEQ") .Date false) "current" (and $current (eq .RelPermalink $current.RelPermalink)) ) -}} {{- end -}} {{- return $entries -}}
