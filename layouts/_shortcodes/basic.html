{{- /* basic.html — a BASIC listing with line numbers. {{< basic >}} 10 PRINT "HELLO" 20 GOTO 10 {{< /basic >}} The line number is split out so it can sit in its own column, and keywords and strings get the two colours a listing on a C64 ever had. The inner text is escaped by the template, so a listing can contain anything. */ -}} {{- $keywords := `(?i)\b(PRINT|GOTO|GOSUB|RETURN|IF|THEN|ELSE|FOR|TO|STEP|NEXT|POKE|PEEK|SYS|LOAD|SAVE|RUN|LIST|NEW|END|STOP|INPUT|GET|DATA|READ|DIM|LET|REM|AND|OR|NOT|CHR\$|RND|LEN|VAL|STR\$|ASC|TAB|SPC|OPEN|CLOSE|CLR|CONT|DEF|FN|ON|WAIT|VERIFY)\b` -}} 
{{- range (.Inner | strings.TrimSpace | split "\n") -}}
{{- $line := . | strings.TrimRight "\r" -}}
{{- $m := findRESubmatch `^(\s*)(\d+)(\s*)(.*)$` $line 1 -}}

{{- if $m -}}
  {{- $parts := index $m 0 -}}
  {{ index $parts 2 }}
  {{- $code := index $parts 4 -}}
  {{- if hasPrefix (upper (strings.TrimSpace $code)) "REM" -}}
    {{ $code }}
  {{- else -}}
    {{- $code = $code | replaceRE `"([^"]*)"` `"$1"` -}}
    {{- $code = $code | replaceRE $keywords `$1` -}}
    {{- $code | safeHTML -}}
  {{- end -}}
{{- else -}}
  {{- $line -}}
{{- end -}}

{{- end -}}
